function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var tasks = [];
  
  if(rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  
  for (var i = 1; i < rows.length; i++) {
    tasks.push({
      id: rows[i][0],
      title: rows[i][1],
      priority: rows[i][2],
      duration: rows[i][3],
      category: rows[i][4],
      targetDate: rows[i][5],
      completed: rows[i][6] === true || rows[i][6] === "true"
    });
  }
  return ContentService.createTextOutput(JSON.stringify(tasks))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  var action = data.action;
  
  if (sheet.getLastRow() == 0) {
    sheet.appendRow(["ID", "Title", "Priority", "Duration", "Category", "TargetDate", "Completed"]);
  }
  
  var rows = sheet.getDataRange().getValues();
  
  if (action === "create") {
    sheet.appendRow([data.id, data.title, data.priority, data.duration, data.category, data.targetDate, data.completed]);
    return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "update") {
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.id) {
        sheet.getRange(i + 1, 2).setValue(data.title);
        sheet.getRange(i + 1, 3).setValue(data.priority);
        sheet.getRange(i + 1, 4).setValue(data.duration);
        sheet.getRange(i + 1, 5).setValue(data.category);
        sheet.getRange(i + 1, 6).setValue(data.targetDate);
        sheet.getRange(i + 1, 7).setValue(data.completed);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "delete") {
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.id) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
  }
}
